##### 3.安全组

安全组是云平台提供的一种虚拟防火墙，用来控制一台或多台云服务器的入站和出站流量。你可以把它理解成一套“允许/拒绝哪些 IP、哪些端口能访问这台机器”的规则集合。普通安全组的默认组内联通策略为**组内互通**，可修改组内联通策略。

```
ssh-copy-id root@172.17.163.238

```

```bash
# 在 master01 上
chmod 600 /root/.ssh/id_rsa

# 验证免密通了再跑脚本
ssh -o StrictHostKeyChecking=no root@172.17.163.241 hostname
ssh -o StrictHostKeyChecking=no root@172.17.163.240 hostname

-------------------------------------
# 用法：在 master01 上执行，前两个参数是两台 worker 的内网 IP

# 第 3 个参数可选：root 密码（不传则用 ~/.ssh/id_rsa 密钥免密登录）

chmod +x k8s-cluster-up.sh

```

**完整脚本如下**（复制到 master01 保存为 k8s-cluster-up.sh）：

```bash
#!/usr/bin/env bash
# ============================================================
#  云岚到家课程 - K8s 集群一键部署脚本
#
#  运行位置：master01 节点上执行
#
#  用法：
#    ./k8s-cluster-up.sh <worker01内网IP> <worker02内网IP> [root密码]
#
#  示例：
#    ./k8s-cluster-up.sh 172.17.163.238 172.17.163.237 123456
#
#  注意：
#    - worker 参数是【内网 IP】：master 通过内网免密 ssh 到 worker
#      （安全组需放行内网互访 172.17.0.0/16；公网 SSH 只对管理机开放）
#    - master01 的内网 IP 由脚本自动探测，无需传入
#
#  说明：
#    - 内网 IP 自动探测，公网 IP 通过命令行参数传入
#    - worker 登录方式：
#        * 传了第 3 个参数(root密码)  -> 自动装 sshpass 走密码登录
#        * 未传密码参数              -> 用 ~/.ssh/id_rsa 密钥免密登录
#    - 可重复执行（幂等）：已初始化的集群会先 kubeadm reset 再重建
# ============================================================

set -e
set -o pipefail

# ---------- 1. 参数（worker 内网 IP）----------
W1_IP=$1
W2_IP=$2
ROOT_PASS=$3

if [ -z "$W1_IP" ] || [ -z "$W2_IP" ]; then
  echo "用法: $0 <worker01内网IP> <worker02内网IP> [root密码]"
  echo "示例: $0 172.17.163.238 172.17.163.237 123456"
  exit 1
fi

# ---------- 2. 常量 ----------
MASTER_HOST=master01.itcast.cn
W1_HOST=worker01.itcast.cn
W2_HOST=worker02.itcast.cn

K8S_VERSION=v1.34.11
POD_CIDR=192.168.0.0/16
IMAGE_REPO=registry.aliyuncs.com/google_containers

CALICO_YAML=/root/calico-3.31.5.yaml
SSH_KEY=/root/.ssh/id_rsa

# ---------- 3. 工具函数 ----------
# 远程在 worker 上执行命令（自动选密码/密钥登录）
ssh_w() {
  local ip=$1; shift
  if [ -n "$ROOT_PASS" ]; then
    sshpass -p "$ROOT_PASS" ssh -o StrictHostKeyChecking=no -o ConnectTimeout=10 root@$ip "$@"
  else
    ssh -o StrictHostKeyChecking=no -o ConnectTimeout=10 -i $SSH_KEY root@$ip "$@"
  fi
}

# 探测本机内网 IP
get_internal_ip() {
  ip route get 1.1.1.1 2>/dev/null | awk '{print $7; exit}'
}

# 打印分节标题
step() { echo -e "\n\033[1;32m========== $1 ==========\033[0m"; }

# 写 hosts 文件（三台同一份）
write_hosts() {
  cat > /etc/hosts <<EOF
127.0.0.1   localhost localhost.localdomain localhost4 localhost4.localdomain4
::1         localhost localhost.localdomain localhost6 localhost6.localdomain6

$MASTER_IP $MASTER_HOST master01
$W1_IP $W1_HOST worker01
$W2_IP $W2_HOST worker02
EOF
}

# ---------- 4. 节点准备脚本（在每台节点上执行同一份）----------
NODE_PREP=/tmp/node-prep.sh
cat > $NODE_PREP <<'PREP'
#!/usr/bin/env bash
set -e
HOSTNAME_NEW=$1

echo "==> [节点准备] $HOSTNAME_NEW"

# 0. 清理旧集群残留（幂等：无论上次跑到哪一步，先恢复干净状态再安装）
echo "==> 清理残留状态"
systemctl stop kubelet 2>/dev/null || true
if command -v kubeadm >/dev/null 2>&1; then
  kubeadm reset -f > /dev/null 2>&1 || true
fi
# 注意：不能 rm -rf /etc/cni/net.d 整个目录！containerd 的 CRI 插件监控该目录，
# 目录被删除会触发 fatal 退出（"cni conf dir is removed"），导致后续 init 连不上 socket。
rm -f /etc/cni/net.d/* 2>/dev/null || true

mkdir -p /etc/cni/net.d
rm -rf /etc/kubernetes /var/lib/kubelet/pki /var/lib/cni /var/lib/calico 2>/dev/null || true

# 1. 修改主机名
hostnamectl set-hostname $HOSTNAME_NEW

# 2. 关闭 SELinux
setenforce 0 || true
sed -i "s/^SELINUX=.*/SELINUX=disabled/" /etc/selinux/config 2>/dev/null || true

# 3. 关闭防火墙
systemctl disable --now firewalld 2>/dev/null || true

# 4. 清空 iptables
iptables -F 2>/dev/null || true
iptables -t nat -F 2>/dev/null || true
iptables -t mangle -F 2>/dev/null || true

# 5. 加载内核模块
modprobe br_netfilter
modprobe overlay
cat > /etc/modules-load.d/k8s.conf <<KEOF
br_netfilter
overlay
KEOF
cat > /etc/sysctl.d/k8s.conf <<KEOF
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
KEOF
sysctl --system > /dev/null

# 6. 关闭交换分区
swapoff -a || true
sed -i "s/.*swap.*/#&/" /etc/fstab

# 7. 安装 containerd（Rocky 10 无 containerd 包，走 docker-ce 源）
if ! command -v containerd >/dev/null 2>&1; then
  dnf install -y yum-utils > /dev/null
  yum-config-manager --add-repo https://mirrors.aliyun.com/docker-ce/linux/centos/docker-ce.repo > /dev/null 2>&1 || true
  dnf install -y containerd.io > /dev/null
fi

# 7.1 配置镜像加速 hosts.toml
mkdir -p /etc/containerd/certs.d/docker.io
mkdir -p /etc/containerd/certs.d/registry.k8s.io

cat > /etc/containerd/certs.d/docker.io/hosts.toml <<KEOF
server = "https://registry-1.docker.io"

[host."https://docker.m.daocloud.io"]
  capabilities = ["pull", "resolve"]
[host."https://docker.mirrors.ustc.edu.cn"]
  capabilities = ["pull", "resolve"]
KEOF

cat > /etc/containerd/certs.d/registry.k8s.io/hosts.toml <<KEOF
server = "https://registry.k8s.io"

[host."https://registry.aliyuncs.com/google_containers"]
  capabilities = ["pull", "resolve"]
KEOF

# 7.2 生成 containerd 默认配置并做 3 处修改（v2.3.4 实测：配合 7.1 的 hosts.toml 镜像加速）
containerd config default > /etc/containerd/config.toml
sed -ri 's#SystemdCgroup = false#SystemdCgroup = true#' /etc/containerd/config.toml
sed -ri 's#registry.k8s.io/pause:3.10.2#registry.aliyuncs.com/google_containers/pause:3.10#' /etc/containerd/config.toml
sed -ri "s#config_path = ''#config_path = '/etc/containerd/certs.d'#" /etc/containerd/config.toml

systemctl daemon-reload

systemctl enable --now containerd
# 强制重启，确保刚写入的 config.toml 生效（幂等重跑时 containerd 可能已在运行）
systemctl restart containerd

# 8. 安装 kubeadm / kubelet / kubectl
if ! command -v kubeadm >/dev/null 2>&1; then
  cat > /etc/yum.repos.d/kubernetes.repo <<KEOF
[kubernetes]
name=Kubernetes
baseurl=https://pkgs.k8s.io/core:/stable:/v1.34/rpm/
enabled=1
gpgcheck=0
gpgkey=https://pkgs.k8s.io/core:/stable:/v1.34/rpm/repodata/repomd.xml.key
KEOF
  dnf install -y kubelet kubeadm kubectl > /dev/null
fi

systemctl enable kubelet

echo "==> [节点准备完成] $HOSTNAME_NEW"
PREP

chmod +x $NODE_PREP

# 若走密码登录，先装 sshpass
if [ -n "$ROOT_PASS" ] && ! command -v sshpass >/dev/null 2>&1; then
  dnf install -y epel-release > /dev/null 2>&1 || true
  dnf install -y sshpass > /dev/null
fi

# ---------- 5. 探测 master 内网 IP ----------
step "1/8 探测 master 内网 IP"
MASTER_IP=$(get_internal_ip)
echo "master01 内网 IP: $MASTER_IP"
echo "worker01 内网 IP: $W1_IP（参数传入）"
echo "worker02 内网 IP: $W2_IP（参数传入）"

# ---------- 6. 写 hosts ----------
step "2/8 写入 /etc/hosts（域名解析）"
write_hosts
ssh_w $W1_IP "cat > /etc/hosts <<EOF
$(cat /etc/hosts)
EOF"
ssh_w $W2_IP "cat > /etc/hosts <<EOF
$(cat /etc/hosts)
EOF"
echo "三台机器 hosts 已写入"
cat /etc/hosts | grep -v "^#"

# ---------- 7. 节点准备（三台都执行）----------
step "3/8 环境准备 + 安装 containerd/kubeadm（三台）"
echo "==> master01 本机执行"
bash $NODE_PREP $MASTER_HOST
echo "==> worker01 远程执行"
scp_pub() { [ -n "$ROOT_PASS" ] && sshpass -p "$ROOT_PASS" scp -o StrictHostKeyChecking=no $NODE_PREP root@$1:/tmp/ || scp -o StrictHostKeyChecking=no -i $SSH_KEY $NODE_PREP root@$1:/tmp/; }
scp_pub $W1_IP
ssh_w $W1_IP "bash /tmp/node-prep.sh $W1_HOST"
echo "==> worker02 远程执行"
scp_pub $W2_IP
ssh_w $W2_IP "bash /tmp/node-prep.sh $W2_HOST"

# ---------- 8. 初始化 Master ----------
step "4/8 初始化 Master 节点（域名初始化）"
# 幂等：若已初始化先 reset
if [ -f /etc/kubernetes/kubelet.conf ]; then
  echo "==> 检测到已初始化的集群，先 kubeadm reset"
  kubeadm reset -f > /dev/null 2>&1 || true
  rm -rf /var/lib/calico
  # 注意：不能 rm -rf /etc/cni/net.d 整个目录！containerd 的 CRI 插件会监控该目录，
  # 目录被删除会触发 fatal 退出（"cni conf dir is removed"），导致后续 init 连不上 socket。
  # 只清空里面的配置文件，保留目录本身。
  rm -f /etc/cni/net.d/* 2>/dev/null || true
  iptables -F 2>/dev/null || true
  iptables -t nat -F 2>/dev/null || true
  iptables -t mangle -F 2>/dev/null || true
fi

# 兜底：保证 CNI 目录存在（kubeadm init / Calico 都要用）
mkdir -p /etc/cni/net.d

# 确保 containerd 存活且 socket 就绪（reset / 清理 CNI 期间它可能退出过）
systemctl is-active containerd > /dev/null 2>&1 || systemctl start containerd
for i in $(seq 1 15); do
  [ -S /var/run/containerd/containerd.sock ] && break
  sleep 2
done
[ -S /var/run/containerd/containerd.sock ] || { echo "错误: containerd socket 不可用，无法初始化"; exit 1; }
echo "containerd socket 就绪"

kubeadm init \
  --kubernetes-version=$K8S_VERSION \
  --image-repository=$IMAGE_REPO \
  --apiserver-advertise-address=$MASTER_IP \
  --control-plane-endpoint=$MASTER_HOST:6443 \
  --pod-network-cidr=$POD_CIDR 2>&1 | tee /root/kubeadm-init.log

# ---------- 9. 配置 kubectl ----------
step "5/8 配置 kubectl"
mkdir -p $HOME/.kube
rm -f $HOME/.kube/config
cp /etc/kubernetes/admin.conf $HOME/.kube/config
chown $(id -u):$(id -g) $HOME/.kube/config
echo "kubeconfig server: $(grep 'server:' $HOME/.kube/config)"

# ---------- 10. 安装 Calico ----------
step "6/8 安装 Calico 网络插件"
if [ ! -f $CALICO_YAML ]; then
  echo "错误: 找不到 $CALICO_YAML，请先从讲义 4.4.1 附件上传 calico-3.31.5.yaml"
  exit 1
fi
kubectl apply -f $CALICO_YAML

# 等 master Ready
echo "==> 等待 master 节点 Ready..."
for i in $(seq 1 30); do
  st=$(kubectl get node $MASTER_HOST -o jsonpath="{.status.conditions[?(@.type=='Ready')].status}" 2>/dev/null || true)
  [ "$st" = "True" ] && { echo "master Ready"; break; }
  sleep 5
done

# ---------- 11. 生成并分发 join 命令 ----------
step "7/8 生成 join 命令并分发到 worker"
JOIN_CMD=$(kubeadm token create --print-join-command)
echo "join 命令: $JOIN_CMD"

for w in $W1_IP $W2_IP; do
  echo "==> worker $w 加入集群"
  # 先清理 worker 上的旧集群残留（幂等重建时很关键）：
  #   - /etc/kubernetes 旧配置
  #   - /var/lib/kubelet/pki 旧集群 CA 签发的 kubelet 证书（不清理会导致向新
  #     apiserver 注册时 Unauthorized，无法触发 TLS bootstrap）
  #   - /var/lib/cni 旧 CNI 状态
  # 用 kubeadm reset 兜底（会停 kubelet、删 pod），再清 pki/cni 残留
  ssh_w $w "kubeadm reset -f > /dev/null 2>&1 || true; \
    rm -rf /var/lib/kubelet/pki /var/lib/cni /etc/cni/net.d/* 2>/dev/null; \
    $JOIN_CMD" 2>&1 | tail -3
done

# ---------- 12. 验证 ----------
step "8/8 验证集群"
export KUBECONFIG=$HOME/.kube/config
for i in $(seq 1 30); do
  # 用 awk 精确匹配 STATUS 列为 Ready（grep -c "Ready" 会把 "NotReady" 也数进去）
  ready=$(kubectl get node --no-headers 2>/dev/null | awk '$2=="Ready"{c++} END{print c+0}')
  [ "$ready" -ge 3 ] && { echo "全部节点 Ready"; break; }
  sleep 5
done
kubectl get node -o wide
kubectl cluster-info | head -1
echo -e "\n\033[1;32m========== 集群部署完成 ==========\033[0m"
echo "控制面地址: $(kubectl cluster-info | grep -oP 'https://[^ ]+')"
```
